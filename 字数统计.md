
```dataviewjs
const LIMIT = 300;               // 字数阈值
const EXCLUDE_TAG_PREFIX = "#video";
const EXCLUDE_FOLDER = "utils/";

let pages = dv.pages("")
    .where(p =>
        p.file &&
        p.file.path &&
        // 排除 video 标签
        //!(p.file.tags ?? []).some(t => t.startsWith(EXCLUDE_TAG_PREFIX)) &&
        // 排除 utils 文件夹
        !p.file.path.startsWith(EXCLUDE_FOLDER)
    );

let results = [];

for (let page of pages) {
    let content = await dv.io.load(page.file.path);
    if (!content) continue;

    // 去掉 YAML frontmatter
    content = content.replace(/^---[\s\S]*?---/, "");

    // 去掉所有空白字符
    let text = content.replace(/\s+/g, "");

    let count = text.length;

    if (count < LIMIT) {
        results.push({
            link: page.file.link,
            words: count
        });
    }
}

results.sort((a, b) => a.words - b.words);

dv.table(
    ["笔记", "字数"],
    results.map(r => [r.link, r.words])
);
```
